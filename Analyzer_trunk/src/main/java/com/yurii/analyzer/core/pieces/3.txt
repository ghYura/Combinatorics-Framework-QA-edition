// ── 3.txt — initial implementation of compressSignature(String) ──────
// Contract:
//   in   String text
//   out  signature string built from category run-lengths collapsed:
//          D = digit, L = letter, S = whitespace, P = punct
//        ("123 abc!" → "DSLP")
//
// Original AnalyzerCore.java: char-by-char with prev-char comparison.

    public static String compressSignature(String text) {
        long __t0 = System.nanoTime();
        Metrics.p3_invocations++;
        Metrics.p3_total_input_chars += (text == null ? 0 : text.length());
        try {
            if (text == null || text.isEmpty()) return "";
            StringBuilder sb = new StringBuilder();
            char prev = 0;
            for (int i = 0; i < text.length(); i++) {
                char ch = text.charAt(i);
                char cur = Character.isDigit(ch) ? 'D'
                         : Character.isLetter(ch) ? 'L'
                         : Character.isWhitespace(ch) ? 'S' : 'P';
                if (cur != prev) { sb.append(cur); prev = cur; }
            }
            String out = sb.toString();
            Metrics.p3_total_output_chars += out.length();
            return out;
        } finally {
            Metrics.p3_duration_ns += System.nanoTime() - __t0;
        }
    }
