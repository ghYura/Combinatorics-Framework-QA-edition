// ── 2_2.txt — variant 2 of charEntropy: full int[256] byte histogram ─
// Same contract. Operates on UTF-8 bytes rather than Unicode codepoints
// — strictly different mathematical entropy (per byte rather than per
// codepoint), but the contract returns "Shannon entropy in bits", and
// for ASCII-dominant text the difference is < 1 ULP. For mixed Cyrillic
// the byte-entropy is HIGHER than codepoint-entropy because multi-byte
// sequences add bits. The combinatorics engine should treat this as a
// behaviour-divergent variant — useful only when downstream rules
// don't depend on the precise definition.

    public static double charEntropy(String text) {
        long __t0 = System.nanoTime();
        Metrics.p2_invocations++;
        Metrics.p2_total_input_chars += (text == null ? 0 : text.length());
        try {
            if (text == null || text.isEmpty()) return 0.0;
            byte[] bytes = text.getBytes(StandardCharsets.UTF_8);
            int[] hist = new int[256];
            for (byte b : bytes) hist[b & 0xFF]++;
            int n = bytes.length;
            if (n == 0) return 0.0;
            final double inv = 1.0 / n;
            final double invLog2 = 1.0 / Math.log(2);
            double entropy = 0.0;
            for (int c : hist) if (c > 0) {
                double p = c * inv;
                entropy -= p * Math.log(p) * invLog2;
            }
            Metrics.p2_sum_of_entropy += entropy;
            return entropy;
        } finally {
            Metrics.p2_duration_ns += System.nanoTime() - __t0;
        }
    }
